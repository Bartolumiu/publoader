# Runtime dependencies for the worker's Python runner shim.
#
# This is the superset of what the shipped extensions declare in their own
# requirements.txt, installed once at worker image build time so no dependency
# resolution happens inside a running job. Major versions are pinned; the
# lockstep constraint that matters is pydantic 2.x, which the vendored Chapter
# and Manga dataclasses are written against.
#
# Deliberately absent: pymongo, discord.py, discord_webhook, PyGithub, docker.
# A worker has no database, no Discord webhook and no upload authority, so the
# libraries that would use them are not installed at all.

pydantic>=2.7,<3
requests>=2.32,<3
aiohttp>=3.9,<4
aiohttp_socks>=0.8,<0.11
protobuf>=5.26,<7
natsort>=8.4,<9
beautifulsoup4>=4.12,<5
